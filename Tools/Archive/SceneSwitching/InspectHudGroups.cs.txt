using System;using System.IO;using System.Linq;using UnityEngine;using UnityEngine.UI;using UnityEditor;using UnityEditor.SceneManagement;using TMPro;using EternalSteam.OpenWorld;using Newtonsoft.Json.Linq;
public static class InspectHudGroups{
public static string Main(){if(EditorApplication.isPlaying||EditorSceneManager.GetActiveScene().isDirty)throw new Exception("Clean edit mode required");var setup=EditorSceneManager.GetSceneManagerSetup();var list=new JArray();try{foreach(var n in new[]{"StartRegionSandbox","OpenWorldSandbox"}){EditorSceneManager.OpenScene("Assets/EternalSteam/Scene/Tests/"+n+".unity");var h=UnityEngine.Object.FindFirstObjectByType<CanvasWorldHud>();var r=h.Sandbox.RailwayHud;string Path(Transform t)=>t.parent==null?t.name:Path(t.parent)+"/"+t.name;var item=new JObject{{"scene",n},{"hud",Path(h.transform)},{"rail",Path(r.transform)},{"sections",new JArray(h.Sections.Select(s=>new JObject{{"id",s.Id},{"path",Path(s.Body.transform)}}))},{"commands",new JArray(r.Commands.Select(c=>new JObject{{"command",c.Command},{"path",Path(c.Button.transform)},{"label",c.Button.GetComponentInChildren<TMP_Text>()?.text}}))},{"hudButtons",new JArray(h.Buttons.Select(b=>new JObject{{"id",b.Id},{"path",Path(b.View.transform)}}))},{"nodes",new JArray(h.GetComponentsInChildren<RectTransform>(true).Where(t=>t.GetComponent<Image>()!=null||t.name is "LeftStatus" or "RightStatus" or "TopBar").Select(t=>new JObject{{"path",Path(t)},{"rect",t.rect.ToString()},{"position",t.anchoredPosition.ToString()}}))}};list.Add(item);}}finally{EditorSceneManager.RestoreSceneManagerSetup(setup);}File.WriteAllText("Docs/Measurements/2026-10-02-hud-groups/inventory.json",list.ToString());return "Inventory saved for both scenes";}}
